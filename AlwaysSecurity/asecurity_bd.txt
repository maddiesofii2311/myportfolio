select utilizadores.login, logs.dat_oper, operacoes.desc_oper
from utilizadores, logs, operacoes
where utilizadores.cod_ut = logs.cod_ut and operacoes.cod_oper = logs.cod_oper;

select utilizadores.login, topicos.msg_top, topicos.cod_ut_top
from utilizadores, topicos
where utilizadores.cod_ut = topicos.cod_ut_top;

select utilizadores.login, comunicacoes.msg_com
from utilizadores, comunicacoes
where utilizadores.cod_ut = comunicacoes.cod_ut_resp and comunicacoes.cod_top = 2;
